select user_id, count(*) as win_count from battle_log where result_win = true group by user_id;

select users.job_id, count(*) as win_count from battle_log join users on battle_log.user_id = users.id where battle_log.result_win = true group by users.job_id;

select level_map,count(id) from(select b.id,
case
when u.level between 0 and 9 then 1
when u.level between 10 and 19 then 10
when u.level between 20 and 29 then 20
when u.level between 30 and 39 then 30
when u.level between 40 and 49 then 40
else 0
end as level_map from battle_log b left outer join users u on b.user_id = u.id where result_win = true ) as tmp group by level_map;

// 例題(さらに難しい)
select level_map as WinCount,count(id) as Num from(
select
b.id,

else 0
end as level_map

left outer join users u on b.user_id = u.id where result_win = true
) as tmp group by level_map;
